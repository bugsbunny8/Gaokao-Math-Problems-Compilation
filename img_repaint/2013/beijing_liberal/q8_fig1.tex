\begin{tikzpicture}[>=Stealth, scale=1.1, line cap=round, line join=round]
  % Coordinates of the cube
  \coordinate (A) at (0, 0);
  \coordinate (B) at (2.4, 0);
  \coordinate (D) at (0.8, 0.8);
  \coordinate (C) at (3.2, 0.8);
